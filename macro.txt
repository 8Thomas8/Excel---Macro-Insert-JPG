Public Sub InsertImage()
    Dim cell As Range
    Dim imageName As String
    Dim imagePath As String
    Dim img As Picture

    On Error GoTo ErrorHandler

    For Each cell In Selection
        ' Définir le nom et le chemin de l'image
        imageName = cell.Offset(0, -1).Value & ".jpg"
        imagePath = ThisWorkbook.Path & "\" & imageName

        ' Vérifier si le fichier image existe
        If Dir(imagePath) <> "" Then
            ' Insérer l'image
            Set img = ActiveSheet.Pictures.Insert(imagePath)
            img.Select

            With Selection.ShapeRange
                .LockAspectRatio = msoTrue
                .Top = cell.Top
                .Left = cell.Left
                .Height = cell.Height
                .IncrementLeft (cell.Width - .Width) / 2
                .IncrementTop (cell.Height - .Height) / 2
            End With

            With Selection
                .PrintObject = True
                .Placement = xlMoveAndSize
            End With
        Else
            MsgBox "Le fichier " & imagePath & " n'existe pas.", vbExclamation
        End If
    Next cell

    Exit Sub

ErrorHandler:
    MsgBox "Une erreur est survenue: " & Err.Description, vbCritical
End Sub
